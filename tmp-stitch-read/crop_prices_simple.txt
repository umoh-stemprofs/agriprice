<script>
  (function initSimpleMarketView() {
    // Refresh button feedback
    const refreshBtn = document.getElementById('refreshFeedBtn');
    if (refreshBtn) {
      refreshBtn.addEventListener('click', () => {
        const icon = refreshBtn.querySelector('.material-symbols-outlined');
        if (icon) {
          icon.classList.add('animate-spin');
          setTimeout(() => {
            icon.classList.remove('animate-spin');
          }, 600);
        }
      });
    }

    // Interactive View Switcher for Price Movement Container
    const switcher = document.getElementById('chartViewSwitcher');
    if (switcher) {
      const buttons = switcher.querySelectorAll('.view-btn');
      const panels = {
        line: document.getElementById('view-line'),
        bar: document.getElementById('view-bar'),
        table: document.getElementById('view-table'),
        pie: document.getElementById('view-pie')
      };

      buttons.forEach(btn => {
        btn.addEventListener('click', () => {
          const targetView = btn.getAttribute('data-view');
          
          // Update button styles
          buttons.forEach(b => {
            b.classList.remove('bg-primary-container', 'text-white', 'shadow-sm', 'font-bold');
            b.classList.add('text-secondary', 'font-semibold');
            b.setAttribute('aria-selected', 'false');
          });

          btn.classList.add('bg-primary-container', 'text-white', 'shadow-sm', 'font-bold');
          btn.classList.remove('text-secondary', 'font-semibold');
          btn.setAttribute('aria-selected', 'true');

          // Switch panel visibility
          Object.keys(panels).forEach(key => {
            if (panels[key]) {
              if (key === targetView) {
                panels[key].classList.remove('hidden');
                panels[key].classList.add('block');
              } else {
                panels[key].classList.add('hidden');
                panels[key].classList.remove('block');
              }
            }
          });
        });
      });
    }
  })();
</script>
