<script>
  // Simple Currency Switcher Logic
  const currencyBtns = document.querySelectorAll('.curr-btn');
  const topTotalDisplay = document.getElementById('topTotalDisplay');
  const topCurrencyCode = document.getElementById('topCurrencyCode');

  currencyBtns.forEach(btn => {
    btn.addEventListener('click', () => {
      // Style active state
      currencyBtns.forEach(b => {
        b.classList.remove('border-primary', 'bg-emerald-50/60');
        b.classList.add('border-[#E2E8F0]', 'bg-white');
        const label = b.querySelector('span:first-child');
        if (label) {
          label.classList.remove('text-primary');
          label.classList.add('text-secondary');
        }
      });

      btn.classList.remove('border-[#E2E8F0]', 'bg-white');
      btn.classList.add('border-primary', 'bg-emerald-50/60');
      const activeLabel = btn.querySelector('span:first-child');
      if (activeLabel) {
        activeLabel.classList.remove('text-secondary');
        activeLabel.classList.add('text-primary');
      }

      // Update Top Massive Total
      topTotalDisplay.textContent = btn.getAttribute('data-total');
      topCurrencyCode.textContent = btn.getAttribute('data-unit');
    });
  });
</script>
